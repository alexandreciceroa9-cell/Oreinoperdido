Munição:", arma.municao ); return true; } // recarregar function recarregar(){ arma.municao = arma.maxMunicao; console.log( "Arma recarregada!" ); } // melhorar arma function melhorarArma(){ arma.nivel++; arma.dano +=10; arma.maxMunicao +=5; arma.municao = arma.maxMunicao; localStorage.setItem( "armaNivel", arma.nivel ); console.log( "Arma melhorada para nível", arma.nivel ); } // carregar melhorias salvas function carregarArma(){ let nivel = localStorage.getItem("armaNivel"); if(nivel){ arma.nivel = Number(nivel); arma.dano = 20 + ((arma.nivel-1)*10); arma.maxMunicao = 10 + ((arma.nivel-1)*5); arma.municao = arma.maxMunicao; } }
